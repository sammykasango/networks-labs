!  FILE       : 03_ACCESS-SW-POSTGRAD-ROOM_VLAN30.txt
!  DEVICE     : Access Switch — Postgraduate Research Room
!  LAYER      : Access Layer
!  VLAN       : VLAN 30 (POSTGRAD)
!  SUBNET     : 192.168.50.96/28
!  HOSTS      : Up to 14 end devices (inc. Wireless AP)
!  GATEWAY    : 192.168.50.97  (Router subinterface G0/0.30)
! ============================================================

! ------------------------------------------------------------
! STEP 1: Enter Privileged EXEC Mode
! ------------------------------------------------------------
enable

! ------------------------------------------------------------
! STEP 2: Enter Global Configuration Mode
! ------------------------------------------------------------
configure terminal

! ------------------------------------------------------------
! STEP 3: Set Device Hostname
! ------------------------------------------------------------
hostname SW-POSTGRAD-ROOM

! ------------------------------------------------------------
! STEP 4: Disable IP Domain Lookup
! ------------------------------------------------------------
no ip domain-lookup

! ------------------------------------------------------------
! STEP 5: Create VLAN 30 and Name It
! ------------------------------------------------------------
vlan 30
 name POSTGRAD
exit

! ------------------------------------------------------------
! STEP 6: Configure Access Ports for Postgrad PCs
!         (Wired end devices)
! ------------------------------------------------------------
interface range FastEthernet 0/1 - 12
 description PC-POSTGRAD-WIRED
 switchport mode access
 switchport access vlan 30
 spanning-tree portfast
 no shutdown
exit

! ------------------------------------------------------------
! STEP 7: Configure Port for Wireless Access Point
!         (AP bridges wireless clients onto VLAN 30)
! ------------------------------------------------------------
interface FastEthernet 0/13
 description WIRELESS-ACCESS-POINT
 switchport mode access
 switchport access vlan 30
 spanning-tree portfast
 no shutdown
exit

! ------------------------------------------------------------
! STEP 8: Configure Uplink Trunk Port to Core Switch
! ------------------------------------------------------------
interface GigabitEthernet 0/1
 description UPLINK-TO-CORE-SWITCH
 switchport mode trunk
 switchport trunk allowed vlan 30
 no shutdown
exit

! ------------------------------------------------------------
! STEP 9: Configure Management VLAN Interface
! ------------------------------------------------------------
interface vlan 30
 description MANAGEMENT-POSTGRAD-ROOM
 ip address 192.168.50.100 255.255.255.240
 no shutdown
exit

ip default-gateway 192.168.50.97

! ------------------------------------------------------------
! STEP 10: Save Configuration
! ------------------------------------------------------------
end
write memory

! ============================================================
!  END OF CONFIGURATION — SW-POSTGRAD-ROOM
! ============================================================
